        private static void EnsureTagsExist()
        {
            SerializedObject tagManager = new SerializedObject(AssetDatabase.LoadAllAssetsAtPath("ProjectSettings/TagManager.asset")[0]);
            SerializedProperty tagsProp = tagManager.FindProperty("tags");

            string[] requiredTags = { "Enemy", "Base", "Projectile", "DropItem", "Defender", "Player" };
            foreach (string t in requiredTags)
            {
                bool found = false;
                for (int i = 0; i < tagsProp.arraySize; i++)
                {
                    SerializedProperty p = tagsProp.GetArrayElementAtIndex(i);
                    if (p.stringValue.Equals(t)) { found = true; break; }
                }
                if (!found)
                {
                    tagsProp.InsertArrayElementAtIndex(tagsProp.arraySize);
                    tagsProp.GetArrayElementAtIndex(tagsProp.arraySize - 1).stringValue = t;
                }
            }
            tagManager.ApplyModifiedProperties();
        }

        private static void ConfigureImportedSprites()
        {
            string[] paths = {
                "Assets/Sprites/Background.jpg",
                "Assets/Sprites/DragonBallAltar.png",
                "Assets/Sprites/Goku.png"
            };

            foreach (string path in paths)
            {
                if (File.Exists(path))
                {
                    TextureImporter importer = AssetImporter.GetAtPath(path) as TextureImporter;
                    if (importer != null)
                    {
                        importer.textureType = TextureImporterType.Sprite;
                        importer.spritePixelsPerUnit = 100;
                        importer.alphaIsTransparency = true;
                        importer.SaveAndReimport();
                    }
                }
            }
        }

        private static void CreateProceduralTextureIfMissing(string path, int width, int height, System.Func<int, int, Color> pixelFunc)
        {
            if (File.Exists(path)) return;

            Texture2D tex = new Texture2D(width, height, TextureFormat.RGBA32, false);
            for (int y = 0; y < height; y++)
            {
                for (int x = 0; x < width; x++)
                {
                    tex.SetPixel(x, y, pixelFunc(x, y));
                }
            }
            tex.Apply();
            byte[] bytes = ImageConversion.EncodeToPNG(tex);
            File.WriteAllBytes(path, bytes);

            AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceUpdate);
            TextureImporter importer = AssetImporter.GetAtPath(path) as TextureImporter;
            if (importer != null)
            {
                importer.textureType = TextureImporterType.Sprite;
                importer.spritePixelsPerUnit = 32;
                importer.filterMode = FilterMode.Point; // Crisp pixel art
                importer.alphaIsTransparency = true;
                importer.SaveAndReimport();
            }
        }

        private static void BuildGameScene(GameObject[] prefabs)
        {

            GameObject camGo = new GameObject("Main Camera");
            Camera cam = camGo.AddComponent<Camera>();
            cam.orthographic = true;
            cam.orthographicSize = 5.5f;
            cam.backgroundColor = new Color(0.08f, 0.08f, 0.12f, 1f);
            camGo.transform.position = new Vector3(0, 0, -10f);
            camGo.AddComponent<ScreenShake>();
            camGo.AddComponent<AudioListener>();
            camGo.tag = "MainCamera";

            // 2. Background Scenery
            GameObject bgGo = new GameObject("Environment_Background");
            SpriteRenderer bgSr = bgGo.AddComponent<SpriteRenderer>();
            Sprite bgSprite = AssetDatabase.LoadAssetAtPath<Sprite>("Assets/Sprites/Background.jpg");
            if (bgSprite != null)
            {
                bgSr.sprite = bgSprite;
                bgSr.sortingOrder = -10;
                // Scale to cover 16:9 11-unit camera
                bgGo.transform.localScale = new Vector3(1.15f, 1.15f, 1f);
                bgGo.transform.position = new Vector3(0.5f, 0, 0);
            }

            // 3. Dragon Ball Altar (Left side base)
            GameObject altarGo = new GameObject("DragonBallAltar");
            altarGo.transform.position = new Vector3(-7.2f, 0, 0);
            SpriteRenderer altarSr = altarGo.AddComponent<SpriteRenderer>();
            Sprite altarSprite = AssetDatabase.LoadAssetAtPath<Sprite>("Assets/Sprites/DragonBallAltar.png");
            if (altarSprite != null)
            {
                altarSr.sprite = altarSprite;
                altarSr.sortingOrder = 2;
                altarGo.transform.localScale = new Vector3(0.38f, 0.38f, 1f);
            }
            BoxCollider2D altarCol = altarGo.AddComponent<BoxCollider2D>();
            altarCol.isTrigger = true;
            altarCol.size = new Vector2(2.5f, 4.5f);
            DragonBallAltar altar = altarGo.AddComponent<DragonBallAltar>();

using System.Collections;
using System.Collections.Generic;
using UnityEngine;

namespace DragonBallDefenders
{
    [RequireComponent(typeof(BoxCollider2D))]
    public class KamehamehaBeam : MonoBehaviour
    {
        public int tickDamage = 60;
        public float duration = 1.2f;
        public float damageInterval = 0.12f;

        private float timer = 0f;
        private float tickTimer = 0f;
        private readonly List<EnemyHealth> enemiesInRange = new List<EnemyHealth>();

        private void Awake()
        {
            BoxCollider2D col = GetComponent<BoxCollider2D>();
            col.isTrigger = true;
            gameObject.tag = "Projectile";
        }

        private void Start()
        {
            ScreenShake.Instance?.Shake(duration, 0.25f);
            SoundManager.Instance?.PlayKamehamehaSound();
        }

        private void Update()
        {
            timer += Time.deltaTime;
            tickTimer += Time.deltaTime;

            if (tickTimer >= damageInterval)
            {
                tickTimer = 0f;
                DamageAllEnemies();
            }

            // Pulse beam scale
            float scaleY = 1f + 0.15f * Mathf.Sin(Time.time * 40f);
            transform.localScale = new Vector3(transform.localScale.x, scaleY, 1f);

            if (timer >= duration)
            {
                Destroy(gameObject);
            }
        }

        private void DamageAllEnemies()
        {
            for (int i = enemiesInRange.Count - 1; i >= 0; i--)
            {
                if (enemiesInRange[i] != null && enemiesInRange[i].gameObject.activeInHierarchy)
                {
                    enemiesInRange[i].TakeDamage(tickDamage, true);
                }
                else
                {
                    enemiesInRange.RemoveAt(i);
                }
            }
        }

        private void OnTriggerEnter2D(Collider2D collision)
        {
            if (collision.CompareTag("Enemy"))
            {
                EnemyHealth enemy = collision.GetComponent<EnemyHealth>();
                if (enemy != null && !enemiesInRange.Contains(enemy))
                {
                    enemiesInRange.Add(enemy);
                    enemy.TakeDamage(tickDamage, true);
                }
            }
        }

        private void OnTriggerExit2D(Collider2D collision)
        {
            if (collision.CompareTag("Enemy"))
            {
                EnemyHealth enemy = collision.GetComponent<EnemyHealth>();
                if (enemy != null)
                {
                    enemiesInRange.Remove(enemy);
                }
            }
        }
    }
}

using System.Collections;
using UnityEngine;

namespace DragonBallDefenders
{
    public class SpiritBomb : MonoBehaviour
    {
        public int explosionDamage = 450;
        public float blastRadius = 3.5f;
        public float travelSpeed = 8f;

        private Vector3 targetPosition;
        private bool hasDetonated = false;

        public void LaunchTo(Vector3 target)
        {
            targetPosition = target;
            StartCoroutine(TravelToTarget());
        }

        private IEnumerator TravelToTarget()
        {
            Vector3 startPos = transform.position;
            float totalDistance = Vector3.Distance(startPos, targetPosition);
            float duration = totalDistance / travelSpeed;
            float elapsed = 0f;

            while (elapsed < duration)
            {
                elapsed += Time.deltaTime;
                float t = elapsed / duration;

                // Parabolic trajectory
                Vector3 currentPos = Vector3.Lerp(startPos, targetPosition, t);
                currentPos.y += Mathf.Sin(t * Mathf.PI) * 1.5f;
                transform.position = currentPos;

                // Rotate & scale pulsation
                transform.Rotate(0, 0, 360f * Time.deltaTime);

                yield return null;
            }

            Detonate();
        }

        private void Detonate()
        {
            if (hasDetonated) return;
            hasDetonated = true;

            ScreenShake.Instance?.Shake(0.5f, 0.45f);
            SoundManager.Instance?.PlayExplosionSound(1f);

            Collider2D[] colliders = Physics2D.OverlapCircleAll(transform.position, blastRadius);
            foreach (var col in colliders)
            {
                if (col.CompareTag("Enemy"))
                {
                    EnemyHealth health = col.GetComponent<EnemyHealth>();
                    if (health != null)
                    {
                        health.TakeDamage(explosionDamage, true);
                    }
                }
            }

            // Visual explosion ring
            GameObject ring = new GameObject("SpiritBombShockwave");
            ring.transform.position = transform.position;
            SpriteRenderer sr = ring.AddComponent<SpriteRenderer>();
            sr.color = new Color(0.2f, 0.8f, 1f, 0.9f);
            ring.transform.localScale = Vector3.one * 0.5f;

            // Animate expand and fade
            SpiritBombShockwave shock = ring.AddComponent<SpiritBombShockwave>();
            shock.maxScale = blastRadius * 2f;

            Destroy(gameObject);
        }

        private void OnDrawGizmosSelected()
        {
            Gizmos.color = Color.cyan;
            Gizmos.DrawWireSphere(transform.position, blastRadius);
        }
    }

    public class SpiritBombShockwave : MonoBehaviour
    {
        public float maxScale = 7f;
        private float duration = 0.5f;
        private float timer = 0f;
        private SpriteRenderer sr;

        private void Awake()
        {
            sr = GetComponent<SpriteRenderer>();
        }

        private void Update()
        {
            timer += Time.deltaTime;
            float t = timer / duration;
            transform.localScale = Vector3.one * Mathf.Lerp(0.5f, maxScale, t);

            if (sr != null)
            {
                sr.color = new Color(sr.color.r, sr.color.g, sr.color.b, 1f - t);
            }

            if (timer >= duration)
            {
                Destroy(gameObject);
            }
        }
    }
}

        private enum ToneType { SineChirp, SawBuzz, PowerUpSweep, Arpeggio, HealChime, SquareAlarm, VictoryFanfare }

        private AudioClip CreateToneClip(string name, float length, float startFreq, float endFreq, float volume, ToneType type)
        {
            int sampleRate = 44100;
            int sampleCount = Mathf.CeilToInt(sampleRate * length);
            float[] samples = new float[sampleCount];

            for (int i = 0; i < sampleCount; i++)
            {
                float t = (float)i / sampleCount;
                float currentFreq = Mathf.Lerp(startFreq, endFreq, t);
                float envelope = (1f - t); // fade out
                float timeSec = (float)i / sampleRate;

                float sample = 0f;
                switch (type)
                {
                    case ToneType.SineChirp:
                        sample = Mathf.Sin(2f * Mathf.PI * currentFreq * timeSec) * envelope;
                        break;
                    case ToneType.SawBuzz:
                        float phase = (currentFreq * timeSec) % 1f;
                        sample = (2f * phase - 1f) * envelope * (0.8f + 0.2f * Mathf.Sin(20f * Mathf.PI * timeSec));
                        break;
                    case ToneType.PowerUpSweep:
                        float sweepFreq = Mathf.Lerp(startFreq, endFreq, Mathf.Pow(t, 2f));
                        sample = (Mathf.Sin(2f * Mathf.PI * sweepFreq * timeSec) + 0.5f * Mathf.Sin(4f * Mathf.PI * sweepFreq * timeSec)) * envelope;
                        break;
                    case ToneType.Arpeggio:
                        int noteIndex = (int)(t * 4);
                        float arpF = startFreq * Mathf.Pow(1.25f, noteIndex);
                        sample = Mathf.Sin(2f * Mathf.PI * arpF * timeSec) * envelope;
                        break;
                    case ToneType.HealChime:
                        sample = (Mathf.Sin(2f * Mathf.PI * currentFreq * timeSec) + 0.5f * Mathf.Sin(2f * Mathf.PI * (currentFreq * 1.5f) * timeSec)) * Mathf.Sin(Mathf.PI * t);
                        break;
                    case ToneType.SquareAlarm:
                        sample = (Mathf.Sin(2f * Mathf.PI * currentFreq * timeSec) > 0 ? 1f : -1f) * envelope * 0.4f;
                        break;
                    case ToneType.VictoryFanfare:
                        int step = Mathf.FloorToInt(t * 5);
                        float stepFreq = startFreq * (step == 0 ? 1f : (step == 1 ? 1.25f : (step == 2 ? 1.5f : (step == 3 ? 1.75f : 2f))));
                        sample = Mathf.Sin(2f * Mathf.PI * stepFreq * timeSec) * (1f - (t % 0.2f) * 4f);
                        break;
                }

                samples[i] = sample * volume;
            }

            AudioClip clip = AudioClip.Create(name, sampleCount, 1, sampleRate, false);
            clip.SetData(samples, 0);
            return clip;
        }

        private AudioClip CreateNoiseClip(string name, float length, float volume)
        {
            int sampleRate = 44100;
            int sampleCount = Mathf.CeilToInt(sampleRate * length);
            float[] samples = new float[sampleCount];

            for (int i = 0; i < sampleCount; i++)
            {
                float t = (float)i / sampleCount;
                float envelope = Mathf.Exp(-t * 6f);
                float noise = (UnityEngine.Random.value * 2f - 1f);
                samples[i] = noise * envelope * volume;
            }

            AudioClip clip = AudioClip.Create(name, sampleCount, 1, sampleRate, false);
            clip.SetData(samples, 0);
            return clip;
        }

using System.Collections;
using UnityEngine;

namespace DragonBallDefenders
{
    public class ScreenShake : MonoBehaviour
    {
        public static ScreenShake Instance { get; private set; }

        private Vector3 initialPosition;
        private Coroutine currentShakeCoroutine;

        private void Awake()
        {
            if (Instance == null)
            {
                Instance = this;
            }
            else
            {
                Destroy(gameObject);
                return;
            }

            initialPosition = transform.localPosition;
        }

        public void Shake(float duration = 0.2f, float magnitude = 0.2f)
        {
            if (currentShakeCoroutine != null)
            {
                StopCoroutine(currentShakeCoroutine);
            }
            currentShakeCoroutine = StartCoroutine(DoShake(duration, magnitude));
        }

        private IEnumerator DoShake(float duration, float magnitude)
        {
            float elapsed = 0f;

            while (elapsed < duration)
            {
                float x = UnityEngine.Random.Range(-1f, 1f) * magnitude;
                float y = UnityEngine.Random.Range(-1f, 1f) * magnitude;

                transform.localPosition = new Vector3(initialPosition.x + x, initialPosition.y + y, initialPosition.z);

                elapsed += Time.unscaledDeltaTime;

                // Scale pulse

                // Rapid flash between red and yellow
                if (spriteRenderer != null)

    [RequireComponent(typeof(Rigidbody2D))]
    public class EnemyMovement : MonoBehaviour
    {
        [Header("Movement Stats")]
        public float moveSpeed = 2.0f;
        public float speedMultiplier = 1.0f;

        [Header("Waypoints")]
        public Transform[] waypoints;
        private int currentWaypointIndex = 0;

        private Rigidbody2D rb;
        private bool isMoving = true;

        private void Awake()
        {
            rb = GetComponent<Rigidbody2D>();
            rb.bodyType = RigidbodyType2D.Kinematic;
            gameObject.tag = "Enemy";
        }

        public void SetWaypoints(Transform[] points)
        {
            waypoints = points;
            currentWaypointIndex = 0;
            if (waypoints != null && waypoints.Length > 0)
            {
                transform.position = waypoints[0].position;
            }
        }

        public void SetSpeedMultiplier(float mult)
        {
            speedMultiplier = mult;
        }

        public void SetMoving(bool moving)
        {
            isMoving = moving;
        }

        private void Update()
        {
            if (!isMoving || waypoints == null || waypoints.Length == 0) return;

            Transform targetPoint = waypoints[currentWaypointIndex];
            float effectiveSpeed = moveSpeed * speedMultiplier;

            // Move towards waypoint using Vector2.MoveTowards as specified in DOCX
            transform.position = Vector2.MoveTowards(transform.position, targetPoint.position, effectiveSpeed * Time.deltaTime);

            // Reached waypoint?
            if (Vector2.Distance(transform.position, targetPoint.position) < 0.1f)
            {
                currentWaypointIndex++;
                if (currentWaypointIndex >= waypoints.Length)
                {
                    OnReachEndPoint();
                }
            }
        }

        private void OnReachEndPoint()
        {
            // Reached Dragon Ball Altar
            DragonBallAltar altar = DragonBallAltar.Instance;
            if (altar != null)
            {
                EnemyHealth health = GetComponent<EnemyHealth>();
                int dmg = health != null ? health.baseDamage : 50;
                altar.TakeDamage(dmg);
            }

            EnemyHealth eh = GetComponent<EnemyHealth>();
            if (eh != null)
            {
                eh.DespawnWithoutDrops();
            }
            else
            {
                Destroy(gameObject);
            }
        }
    }

            if (GameManager.Instance != null && GameManager.Instance.currentState == GameState.WaveBreak)
            {
                countdown -= Time.deltaTime;
                OnCountdownChanged?.Invoke(Mathf.Max(0f, countdown));
            }

        private IEnumerator WaveLoop()
        {
            while (currentWaveIndex < totalWaves)
            {
                // Break period between waves
                if (GameManager.Instance != null)
                {
                    GameManager.Instance.SetGameState(GameState.WaveBreak);
                }

                countdown = timeBetweenWaves;
                while (countdown > 0)
                {
                    yield return null;
                }

                // Start next wave
                currentWaveIndex++;
                OnWaveChanged?.Invoke(currentWaveIndex, totalWaves);

                if (GameManager.Instance != null)
                {
                    GameManager.Instance.SetGameState(GameState.Playing);
                }

                yield return StartCoroutine(SpawnWave(currentWaveIndex));

                // Wait for all spawned enemies in this wave to be eliminated
                while (remainingEnemiesToKill > 0)
                {
                    yield return new WaitForSeconds(0.5f);
                }

                // Wave Cleared bonus
                if (GameManager.Instance != null)
                {
                    GameManager.Instance.AddZeni(50 + currentWaveIndex * 25);
                    GameManager.Instance.AddScore(500 * currentWaveIndex);
                }

                SoundManager.Instance?.PlayCoinSound();
                yield return new WaitForSeconds(1.0f);
            }

            // All waves cleared -> Victory!
            if (GameManager.Instance != null)
            {
                GameManager.Instance.SetGameState(GameState.Victory);
            }

        public IEnumerator SpawnWave(int waveNumber)

            float spdMult = Mathf.Min(1.6f, Mathf.Pow(speedMultiplierPerWave, waveNumber - 1));

            remainingEnemiesToKill = enemyCount;

            // Check if this wave has a boss
            GameObject bossPrefab = GetBossForWave(waveNumber);
            if (bossPrefab != null)

                remainingEnemiesToKill += 1;

            for (int i = 0; i < enemyCount; i++)

                yield return new WaitForSeconds(1.5f);

using UnityEngine;

namespace DragonBallDefenders
{
    public class WaypointPath : MonoBehaviour
    {
        public Transform[] waypoints;
        public Color pathColor = Color.green;

        public Transform[] GetPoints()
        {
            if (waypoints != null && waypoints.Length > 0)
                return waypoints;

            int childCount = transform.childCount;
            waypoints = new Transform[childCount];
            for (int i = 0; i < childCount; i++)
            {
                waypoints[i] = transform.GetChild(i);
            }
            return waypoints;
        }

        private void OnDrawGizmos()
        {
            Transform[] pts = GetPoints();
            if (pts == null || pts.Length < 2) return;

            Gizmos.color = pathColor;
            for (int i = 0; i < pts.Length - 1; i++)
            {
                if (pts[i] != null && pts[i + 1] != null)
                {
                    Gizmos.DrawLine(pts[i].position, pts[i + 1].position);
                    Gizmos.DrawSphere(pts[i].position, 0.15f);
                }
            }
            if (pts[pts.Length - 1] != null)
            {
                Gizmos.DrawSphere(pts[pts.Length - 1].position, 0.2f);
            }
        }
    }
}

        private Color textColor;

        public static void Create(Vector3 position, int damageAmount, bool isCritical = false)
        {
            GameObject popupObj = new GameObject("DamagePopup");
            popupObj.transform.position = position + new Vector3(UnityEngine.Random.Range(-0.3f, 0.3f), UnityEngine.Random.Range(-0.1f, 0.3f), -2f);

            DamagePopup popup = popupObj.AddComponent<DamagePopup>();
            popup.Setup(damageAmount, isCritical);
        }

        public void Setup(int damageAmount, bool isCritical)
        {
            if (textMesh == null)
            {
                textMesh = gameObject.AddComponent<TextMeshPro>();
                textMesh.alignment = TextAlignmentOptions.Center;
                textMesh.fontSize = isCritical ? 6.5f : 4.8f;
                textMesh.sortingOrder = 50;
            }

            textMesh.text = damageAmount.ToString();
            textColor = isCritical ? new Color(1f, 0.8f, 0.1f, 1f) : new Color(1f, 0.3f, 0.2f, 1f);
            textMesh.color = textColor;

            StartCoroutine(FadeAndDestroy());
        }

        private IEnumerator FadeAndDestroy()
        {
            float elapsed = 0f;
            Vector3 startPos = transform.position;

            while (elapsed < disappearTimer)
            {
                elapsed += Time.deltaTime;
                transform.position = startPos + Vector3.up * (moveSpeed * elapsed);

                float alpha = Mathf.Clamp01(1f - (elapsed / disappearTimer));
                textMesh.color = new Color(textColor.r, textColor.g, textColor.b, alpha);

                yield return null;
            }

            Destroy(gameObject);
        }
    }
}

public class PlayerDefender : MonoBehaviour
{

    public float minY = -3.5f;
    public float maxY = 3.5f;

    [Header("Tấn công")]

    {
        anim = GetComponent<Animator>();
    }

    void Update()
    {
        HandleMovement();
        HandleAttack();
    }

    void HandleMovement()
    {
        float verticalInput = 0f;
#if ENABLE_INPUT_SYSTEM
        if (UnityEngine.InputSystem.Keyboard.current != null)
        {
            if (UnityEngine.InputSystem.Keyboard.current.wKey.isPressed || UnityEngine.InputSystem.Keyboard.current.upArrowKey.isPressed)
                verticalInput += 1f;
            if (UnityEngine.InputSystem.Keyboard.current.sKey.isPressed || UnityEngine.InputSystem.Keyboard.current.downArrowKey.isPressed)
                verticalInput -= 1f;
        }
        else
#endif
        {
            try
            {
                verticalInput = Input.GetAxisRaw("Vertical");
            }
            catch (System.InvalidOperationException) { }
        }

        Vector3 newPos = transform.position + new Vector3(0, verticalInput * moveSpeed * Time.deltaTime, 0);
        newPos.y = Mathf.Clamp(newPos.y, minY, maxY);
        transform.position = newPos;
    }

    void HandleAttack()
    {
        bool fire = false;
#if ENABLE_INPUT_SYSTEM
        if (UnityEngine.InputSystem.Keyboard.current != null && UnityEngine.InputSystem.Keyboard.current.spaceKey.isPressed)
            fire = true;
        if (UnityEngine.InputSystem.Mouse.current != null && UnityEngine.InputSystem.Mouse.current.leftButton.isPressed)
            fire = true;
        if (UnityEngine.InputSystem.Keyboard.current == null && UnityEngine.InputSystem.Mouse.current == null)
#endif
        {
            try
            {
                fire = Input.GetKey(KeyCode.Space) || Input.GetMouseButton(0);
            }
            catch (System.InvalidOperationException) { }
        }

        float effectiveFireRate = GameManager.instance != null ? GameManager.instance.fireRate : 0.4f;
        if (fire && Time.time >= nextFireTime)
        {
            nextFireTime = Time.time + effectiveFireRate;
            ShootKiBlast();
        }
    }

    void ShootKiBlast()
    {
        // Kích hoạt Animation vung tay bắn chưởng của Goku
        if (anim != null)
        {
            anim.SetTrigger("Attack");
        }

        Vector3 spawnPos = firePoint != null ? firePoint.position : transform.position + new Vector3(0.6f, 0.1f, 0);
        if (kiBlastPrefab != null)
        {
            Instantiate(kiBlastPrefab, spawnPos, Quaternion.identity);
        }
        else
        {

using System.Collections;
using UnityEngine;

public class WaveSpawner : MonoBehaviour
{
    [Header("Prefab Quái vật")]
    public GameObject normalEnemyPrefab;
    public GameObject bomberEnemyPrefab;

    [Header("Danh sách Điểm đầu & Điểm cuối")]
    public Transform[] spawnPoints;
    public Transform[] endPoints;

    [Header("Cấu hình Đợt tấn công (Waves)")]
    public int currentWave = 1;
    public int baseEnemiesPerWave = 5;
    public float timeBetweenSpawns = 1.2f;
    public float timeBetweenWaves = 4f;

    private void Start()
    {
        StartCoroutine(SpawnWaveRoutine());
    }

    IEnumerator SpawnWaveRoutine()
    {
        while (true)
        {
            if (GameManager.instance != null)
            {
                GameManager.instance.UpdateWaveUI(currentWave);
            }

            // Tăng dần số lượng, máu và tốc độ quái sau mỗi đợt
            int enemyCount = baseEnemiesPerWave + (currentWave - 1) * 3;
            float hpMultiplier = 1f + (currentWave - 1) * 0.35f;
            float speedBonus = (currentWave - 1) * 0.2f;

            for (int i = 0; i < enemyCount; i++)
            {
                SpawnSingleEnemy(hpMultiplier, speedBonus);
                yield return new WaitForSeconds(timeBetweenSpawns);
            }

            // Đợi người chơi tiêu diệt hết quái trong đợt hiện tại
            yield return new WaitUntil(() => GameObject.FindGameObjectsWithTag("Enemy").Length == 0);

            // Nghỉ giữa hiệp để nâng cấp vũ khí trước khi sang đợt tiếp theo
            yield return new WaitForSeconds(timeBetweenWaves);
            currentWave++;
        }
    }

    void SpawnSingleEnemy(float hpMultiplier, float speedBonus)
    {
        if (spawnPoints == null || spawnPoints.Length == 0 || endPoints == null || endPoints.Length == 0) return;

        int laneIndex = Random.Range(0, spawnPoints.Length);
        laneIndex = Mathf.Clamp(laneIndex, 0, endPoints.Length - 1);
        GameObject prefabToSpawn = (Random.value < 0.35f) ? bomberEnemyPrefab : normalEnemyPrefab;

        if (prefabToSpawn != null)
        {
            GameObject enemyObj = Instantiate(prefabToSpawn);
            Enemy enemyScript = enemyObj.GetComponent<Enemy>();

            if (enemyScript != null)
            {
                enemyScript.SetupEnemy(spawnPoints[laneIndex], endPoints[laneIndex], hpMultiplier, speedBonus);
            }
        }
    }
}

        private static AnimationClip CreateSpriteAnimation(string path, Sprite[] frames, float frameDuration, bool isLooping)
        {
            if (frames == null || frames.Length == 0) return null;

            AnimationClip clip = new AnimationClip();
            clip.frameRate = 30;

            EditorCurveBinding binding = new EditorCurveBinding();
            binding.type = typeof(SpriteRenderer);
            binding.path = "";
            binding.propertyName = "m_Sprite";

            ObjectReferenceKeyframe[] keyframes = new ObjectReferenceKeyframe[frames.Length];
            for (int i = 0; i < frames.Length; i++)
            {
                keyframes[i] = new ObjectReferenceKeyframe();
                keyframes[i].time = i * frameDuration;
                keyframes[i].value = frames[i];
            }

            AnimationUtility.SetObjectReferenceCurve(clip, binding, keyframes);

            AnimationClipSettings settings = AnimationUtility.GetAnimationClipSettings(clip);
            settings.loopTime = isLooping;
            AnimationUtility.SetAnimationClipSettings(clip, settings);

            AssetDatabase.CreateAsset(clip, path);
            return clip;
        }

        private static AnimatorController CreatePlayerAnimator(AnimationClip idleClip, AnimationClip attackClip)
        {
            string path = "Assets/Animations/PlayerAnimator.controller";
            AnimatorController controller = AnimatorController.CreateAnimatorControllerAtPath(path);

            controller.AddParameter("Attack", AnimatorControllerParameterType.Trigger);

            AnimatorStateMachine sm = controller.layers[0].stateMachine;

            AnimatorState idleState = sm.AddState("Goku_Idle");
            idleState.motion = idleClip;
            sm.defaultState = idleState;

            AnimatorState attackState = sm.AddState("Goku_Attack");
            attackState.motion = attackClip;

            // Idle -> Attack transition
            AnimatorStateTransition toAttack = idleState.AddTransition(attackState);
            toAttack.hasExitTime = false;
            toAttack.duration = 0f;
            toAttack.AddCondition(AnimatorConditionMode.If, 0, "Attack");

            // Attack -> Idle transition
            AnimatorStateTransition toIdle = attackState.AddTransition(idleState);
            toIdle.hasExitTime = true;
            toIdle.exitTime = 0.85f;
            toIdle.duration = 0.05f;

            return controller;
        }
